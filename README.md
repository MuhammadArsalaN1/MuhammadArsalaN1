script = r'''import json
import os
import re
from datetime import datetime, timezone, timedelta
from urllib.request import Request, urlopen

USERNAME = os.environ.get("GITHUB_USERNAME", "MuhammadArsalaN1")
TOKEN = os.environ["GITHUB_TOKEN"]
README = "README.md"

API = "https://api.github.com"
HEADERS = {
    "Authorization": f"Bearer {TOKEN}",
    "Accept": "application/vnd.github+json",
    "X-GitHub-Api-Version": "2022-11-28",
    "User-Agent": "profile-readme-updater",
}

def get_json(url):
    req = Request(url, headers=HEADERS)
    with urlopen(req, timeout=30) as response:
        return json.loads(response.read().decode("utf-8"))

def graphql(query, variables):
    body = json.dumps({"query": query, "variables": variables}).encode()
    req = Request(
        "https://api.github.com/graphql",
        data=body,
        headers={**HEADERS, "Content-Type": "application/json"},
        method="POST",
    )
    with urlopen(req, timeout=30) as response:
        data = json.loads(response.read().decode("utf-8"))
    if "errors" in data:
        raise RuntimeError(data["errors"])
    return data["data"]

def relative_time(iso):
    if not iso:
        return "—"
    dt = datetime.fromisoformat(iso.replace("Z", "+00:00"))
    delta = datetime.now(timezone.utc) - dt
    seconds = int(delta.total_seconds())

    if seconds < 60:
        return "just now"
    if seconds < 3600:
        return f"{seconds // 60}m ago"
    if seconds < 86400:
        return f"{seconds // 3600}h ago"
    if seconds < 2592000:
        return f"{seconds // 86400}d ago"
    if seconds < 31536000:
        return f"{seconds // 2592000}mo ago"
    return f"{seconds // 31536000}y ago"

def iso_date(iso):
    if not iso:
        return "—"
    return iso[:10]

def md_table(rows, headers):
    out = [
        "| " + " | ".join(headers) + " |",
        "| " + " | ".join("---" for _ in headers) + " |",
    ]
    out.extend("| " + " | ".join(str(x) for x in row) + " |" for row in rows)
    return "\n".join(out)

def replace_section(text, marker, replacement):
    pattern = rf"(<!-- {re.escape(marker)}:START -->).*?(<!-- {re.escape(marker)}:END -->)"
    updated, count = re.subn(
        pattern,
        lambda m: f"{m.group(1)}\n\n{replacement}\n\n{m.group(2)}",
        text,
        flags=re.S,
    )
    if count != 1:
        raise RuntimeError(f"Could not find exactly one section for {marker}")
    return updated

now = datetime.now(timezone.utc)
year_start = datetime(now.year, 1, 1, tzinfo=timezone.utc)
thirty_days = now - timedelta(days=30)
seven_days = now - timedelta(days=7)

user = get_json(f"{API}/users/{USERNAME}")

repos = []
page = 1
while page <= 5:
    batch = get_json(
        f"{API}/users/{USERNAME}/repos?per_page=100&page={page}&sort=updated&direction=desc"
    )
    if not batch:
        break
    repos.extend([r for r in batch if not r.get("fork")])
    if len(batch) < 100:
        break
    page += 1

events = []
page = 1
while page <= 3:
    batch = get_json(
        f"{API}/users/{USERNAME}/events/public?per_page=100&page={page}"
    )
    if not batch:
        break
    events.extend(batch)
    if len(batch) < 100:
        break
    page += 1

def event_date(event):
    return datetime.fromisoformat(event["created_at"].replace("Z", "+00:00"))

recent_events = sorted(events, key=event_date, reverse=True)
last_30 = [e for e in recent_events if event_date(e) >= thirty_days]
last_7 = [e for e in recent_events if event_date(e) >= seven_days]
year_events = [e for e in recent_events if event_date(e) >= year_start]

push_events = [e for e in recent_events if e["type"] in {"PushEvent", "CreateEvent"}]
pr_events = [e for e in recent_events if e["type"] == "PullRequestEvent"]
issue_events = [e for e in recent_events if e["type"] == "IssuesEvent"]
watch_events = [e for e in recent_events if e["type"] == "WatchEvent"]

# Contribution calendar via GitHub GraphQL.
contribution_query = """
query($login:String!, $from:DateTime!, $to:DateTime!) {
  user(login:$login) {
    contributionsCollection(from:$from, to:$to) {
      contributionCalendar {
        totalContributions
        weeks {
          contributionDays {
            contributionCount
            date
          }
        }
      }
    }
  }
}
"""

try:
    contribution_data = graphql(
        contribution_query,
        {
            "login": USERNAME,
            "from": year_start.isoformat(),
            "to": now.isoformat(),
        },
    )
    calendar = contribution_data["user"]["contributionsCollection"]["contributionCalendar"]
    total_year = calendar["totalContributions"]
    days = [
        day
        for week in calendar["weeks"]
        for day in week["contributionDays"]
    ]
    total_30 = sum(
        d["contributionCount"]
        for d in days
        if datetime.fromisoformat(d["date"]).replace(tzinfo=timezone.utc) >= thirty_days
    )
    total_7 = sum(
        d["contributionCount"]
        for d in days
        if datetime.fromisoformat(d["date"]).replace(tzinfo=timezone.utc) >= seven_days
    )
except Exception:
    # Fall back to public event counts if GraphQL is unavailable.
    total_year = len(year_events)
    total_30 = len(last_30)
    total_7 = len(last_7)

stars = sum(r.get("stargazers_count", 0) for r in repos)
active_cutoff = now - timedelta(days=30)
active_repos = [
    r for r in repos
    if r.get("pushed_at") and datetime.fromisoformat(r["pushed_at"].replace("Z", "+00:00")) >= active_cutoff
]

# Human-readable activity.
def describe_event(event):
    etype = event["type"]
    repo = event.get("repo", {}).get("name", "unknown")

    if etype == "PushEvent":
        count = len(event.get("payload", {}).get("commits", []))
        return f"Pushed {count} commit{'s' if count != 1 else ''}"
    if etype == "PullRequestEvent":
        action = event.get("payload", {}).get("action", "updated")
        number = event.get("payload", {}).get("number", "")
        return f"PR #{number} {action}"
    if etype == "IssuesEvent":
        action = event.get("payload", {}).get("action", "updated")
        number = event.get("payload", {}).get("issue", {}).get("number", "")
        return f"Issue #{number} {action}"
    if etype == "CreateEvent":
        ref_type = event.get("payload", {}).get("ref_type", "repository")
        return f"Created {ref_type}"
    if etype == "DeleteEvent":
        ref_type = event.get("payload", {}).get("ref_type", "branch")
        return f"Deleted {ref_type}"
    if etype == "WatchEvent":
        return "Starred repository"
    if etype == "ForkEvent":
        return "Forked repository"
    if etype == "ReleaseEvent":
        return "Published release"
    return etype.replace("Event", "")

activity_rows = []
seen = set()
for event in recent_events:
    repo_name = event.get("repo", {}).get("name", "")
    key = (event["id"], repo_name)
    if key in seen:
        continue
    seen.add(key)
    activity_rows.append([
        relative_time(event["created_at"]),
        describe_event(event),
        f"[{repo_name}](https://github.com/{repo_name})" if repo_name else "—",
    ])
    if len(activity_rows) == 10:
        break

if not activity_rows:
    activity_rows = [["No recent public events", "—", "—"]]

# Current projects: repositories with recent pushes, then recently updated repos.
current = sorted(
    active_repos,
    key=lambda r: r.get("pushed_at") or "",
    reverse=True,
)[:8]

current_rows = []
for r in current:
    current_rows.append([
        f"[{r['name']}]({r['html_url']})",
        r.get("language") or "Multiple",
        "Active",
        relative_time(r.get("pushed_at")),
    ])

if not current_rows:
    current_rows = [["No recent public activity", "—", "—", "—"]]

# Repository table.
project_rows = []
for r in repos[:15]:
    status = "Active" if r in active_repos else "Maintained"
    description = (r.get("description") or "No description provided.").replace("|", "\\|")
    if len(description) > 100:
        description = description[:97] + "..."
    project_rows.append([
        f"[{r['name']}]({r['html_url']})",
        description,
        r.get("language") or "—",
        status,
        relative_time(r.get("updated_at")),
    ])

if not project_rows:
    project_rows = [["No public repositories", "—", "—", "—", "—"]]

last_activity = relative_time(recent_events[0]["created_at"]) if recent_events else "No public activity"

stats_rows = [
    ["Public repositories", user.get("public_repos", 0)],
    ["Contributions this year", total_year],
    ["Contributions last 30 days", total_30],
    ["Commits / pushes in recent activity", len(push_events)],
    ["Pull requests", len(pr_events)],
    ["Issues", len(issue_events)],
    ["Stars earned", stars],
    ["Followers", user.get("followers", 0)],
    ["Last activity", last_activity],
    ["Profile last refreshed", now.strftime("%Y-%m-%d %H:%M UTC")],
]

contribution_rows = [
    ["This year", total_year],
    ["Last 30 days", total_30],
    ["Last 7 days", total_7],
    ["Latest activity", last_activity],
]

with open(README, "r", encoding="utf-8") as f:
    content = f.read()

content = replace_section(
    content,
    "GITHUB_STATS",
    md_table(stats_rows, ["Metric", "Value"]),
)

content = replace_section(
    content,
    "CURRENTLY_BUILDING",
    md_table(current_rows, ["Repository", "Language", "Activity", "Last Updated"]),
)

content = replace_section(
    content,
    "DAILY_ACTIVITY",
    md_table(activity_rows, ["When", "Activity", "Repository"]),
)

content = replace_section(
    content,
    "CONTRIBUTIONS",
    md_table(contribution_rows, ["Period", "Activity"]),
)

content = replace_section(
    content,
    "PROJECTS",
    md_table(project_rows, ["Project", "Description", "Language", "Status", "Updated"]),
)

with open(README, "w", encoding="utf-8") as f:
    f.write(content)

print("README updated successfully.")
'''
